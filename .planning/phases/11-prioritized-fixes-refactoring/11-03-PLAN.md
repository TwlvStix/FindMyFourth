    static Stream<List<GamesRecord>> queryAvailableGames({
      String? courseFilter,
      String? styleFilter,
      DateTime? dateFilter,
    }) {
      Query query = FirebaseFirestore.instance.collection('games')
        .where('is_private', isEqualTo: false)
        .where('status', isEqualTo: 'open');

      if (courseFilter != null) query = query.where('course', isEqualTo: courseFilter);
      if (styleFilter != null) query = query.where('style_game', isEqualTo: styleFilter);
      if (dateFilter != null) query = query.where('date_time', isGreaterThanOrEqualTo: dateFilter);

      return query.snapshots().map((snapshot) =>
        snapshot.docs.map((doc) => GamesRecord.fromSnapshot(doc)).toList()
      );
    }

    static Stream<List<GamesRecord>> queryUserGames(String userId) {
      return FirebaseFirestore.instance.collection('games')
        .where('joined_players', arrayContains: userId)
        .orderBy('date_time', descending: true)
        .snapshots()
        .map((snapshot) => snapshot.docs.map((doc) => GamesRecord.fromSnapshot(doc)).toList());
    }

    static Future<GamesRecord?> getGameById(String gameId) async {
      try {
        final doc = await FirebaseFirestore.instance.collection('games').doc(gameId).get();
        return doc.exists ? GamesRecord.fromSnapshot(doc) : null;
      } on FirebaseException catch (e) {
        debugPrint('GameService.getGameById error: ${e.code} - ${e.message}');
        rethrow;

    **GameProvider** (lib/providers/game_provider.dart):
    ```dart
    class GameProvider extends ChangeNotifier {
      // State
      Map<String, GamesRecord> _gameCache = {};
      Map<String, DateTime> _gameCacheTimestamps = {};
      Duration _cacheTTL = Duration(minutes: 5);
      Map<String, StreamRequestManager<GamesRecord>> _gameStreamManagers = {};

      // Cache getters
      GamesRecord? getCachedGame(String gameId) => _gameCache[gameId];
      bool isGameCacheValid(String gameId) {
        final timestamp = _gameCacheTimestamps[gameId];
        return timestamp != null && DateTime.now().difference(timestamp) < _cacheTTL;
      }

      // Query with caching
      Future<GamesRecord?> getGame(String gameId) async {
        if (isGameCacheValid(gameId)) return getCachedGame(gameId);

        try {
          final game = await GameService.getGameById(gameId);
          if (game != null) {
            _gameCache[gameId] = game;
            _gameCacheTimestamps[gameId] = DateTime.now();
            notifyListeners();
          }
          return game;
        } catch (e) {
          debugPrint('GameProvider.getGame error: $e');
          rethrow;
        }
      }

      // Streams with StreamRequestManager
      Stream<List<GamesRecord>> availableGamesStream({filters}) {
        // Use StreamRequestManager pattern from UserProvider
      }

      // Mutations with cache invalidation
      Future<void> joinGame(String gameId, String userId) async {
        await GameService.joinGame(gameId, userId);
        invalidateGameCache(gameId);
        await getGame(gameId); // Refresh
      }

      // Cache management
      void invalidateGameCache(String gameId) {
        _gameCache.remove(gameId);
        _gameCacheTimestamps.remove(gameId);
        notifyListeners();
      }

      @override
      void dispose() {
        _gameStreamManagers.values.forEach((m) => m.dispose());
        super.dispose();
      }
    }
    ```
